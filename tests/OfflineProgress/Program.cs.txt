using System;
using System.Reflection;
using IdleAi;

class Program
{
    static int checks;
    static void Equal(double actual, double expected, string label)
    {
        if (actual != expected) throw new Exception($"{label}: {actual:R} != {expected:R}");
        checks++;
    }
    static (GameState, EconomyService, OfflineProgressService) Create()
    {
        var state = new GameState([new RoomData("Test", null, null,
            [new MachineData("Test", 100, 1, 1, 100, null)], 0)]);
        var room = new RoomState { Unlocked = true };
        room.Slots.Add(new SlotData { Unlocked = true, MachineLevel = 1 });
        state.RoomStates.Add(room);
        var economy = new EconomyService(state);
        return (state, economy, new OfflineProgressService(state, economy, new LabService(state, economy)));
    }
    static SaveGameData Save(double income, long boost = 0, int version = 21) => new()
    { LastSaveUnix = 1000, IncomePerSecond = income, ShopProductionBoostEndUnix = boost, SaveVersion = version };
    static double Segment(double income, long start, long end, long boost, double factor)
    {
        long seconds = end - start;
        if (seconds <= 0 || income <= 0) return 0;
        long boosted = Math.Min(Math.Max(0, Math.Min(end, boost) - start), seconds);
        return (income * (seconds - boosted) + income * boosted * GameConfig.ShopTemporaryProductionMultiplier) * factor;
    }
    static void Near(double actual, double expected, string label)
    {
        if (Math.Abs(actual - expected) > 1e-8 * Math.Max(1, Math.Abs(expected)))
            throw new Exception($"{label}: {actual:R} != {expected:R}");
        checks++;
    }
    static SlotData Bot(double lifetime, BotRarity rarity = BotRarity.Rare) => new()
    { Unlocked = true, BotRarity = rarity, BotDurabilityInitialized = true,
      BotDurabilitySecondsRemaining = lifetime };
    static void ConfiguredLifetimeTests()
    {
        var rarities = new[] { BotRarity.Common, BotRarity.Rare, BotRarity.Epic, BotRarity.Legendary };
        var hours = new[] { 24, 36, 48, 72 };
        var oldHours = new[] { 6, 12, 24, 48 };
        var multipliers = new[] { 1.0, 1.2, 1.4, 1.5 };
        for (int rarityIndex = 0; rarityIndex < rarities.Length; rarityIndex++)
        {
            var rarity = rarities[rarityIndex]; double lifetime = hours[rarityIndex] * 3600;
            Equal(BotCatalog.GetWorkingLifetimeSeconds(rarity), lifetime, "exact configured lifetime: " + rarity);
            Equal(BotCatalog.GetMultiplier(rarity), multipliers[rarityIndex], "unchanged production multiplier: " + rarity);
            var (state, economy, offline) = Create();
            for (int roomIndex = 1; roomIndex < 4; roomIndex++)
            {
                state.Rooms.Add(state.Rooms[0]);
                var room = new RoomState { Unlocked = true };
                room.Slots.Add(new SlotData { Unlocked = true, MachineLevel = 1 });
                state.RoomStates.Add(room);
            }
            foreach (var room in state.RoomStates)
            {
                var slot = room.Slots[0]; slot.BotRarity = rarity; slot.BotPurchasePrice = 100;
                slot.InitializeBotDurability(); Equal(slot.BotDurabilitySecondsRemaining, lifetime, "full lifetime in Rooms 1–4");
            }
            new ProductionService(state, economy, new PipelineService(state, economy)).Update(60);
            foreach (var room in state.RoomStates)
                Equal(room.Slots[0].BotDurabilitySecondsRemaining, lifetime - 60, "runtime wear in Rooms 1–4");
            var mapper = new GameSaveService(state, economy);
            offline.ApplyFromSave(mapper.CreateSave(1000), 4600);
            foreach (var room in state.RoomStates)
                Equal(room.Slots[0].BotDurabilitySecondsRemaining, lifetime - 3660, "one hour offline wear in Rooms 1–4");
            offline.ApplyFromSave(mapper.CreateSave(4600), 4600 + (long)lifetime);
            state.Tokens = 1000;
            var bots = new BotService(state, economy);
            for (int roomIndex = 0; roomIndex < 4; roomIndex++)
            {
                var slot = state.RoomStates[roomIndex].Slots[0];
                Equal(slot.BotDurabilitySecondsRemaining, 0, "offline clamps at zero");
                Equal(slot.BotBroken && !slot.HasWorkingBot && !slot.IsRunning ? 1 : 0, 1, "broken bot disables automation");
                Equal(bots.GetRepairPrice(slot), 20, "repair price remains 20% of purchase");
                Equal(bots.RepairBot(roomIndex, 0).Changed ? 1 : 0, 1, "repair succeeds in Rooms 1–4");
                Equal(slot.BotDurabilitySecondsRemaining, lifetime, "repair restores new full lifetime");
                Equal(slot.HasWorkingBot && slot.IsRunning ? 1 : 0, 1, "repair resumes automation");
                Equal(bots.RepairBot(roomIndex, 0).Changed ? 1 : 0, 0, "full durability cannot be repaired twice");
            }
            // Prior-release remaining seconds survive main-save JSON/hydration exactly, including broken bots.
            foreach (double remaining in new[] { 0, 1234.5, oldHours[rarityIndex] * 3600.0 })
            {
                foreach (var room in state.RoomStates) room.Slots[0].BotDurabilitySecondsRemaining = remaining;
                var json = System.Text.Json.JsonSerializer.Serialize(mapper.CreateSave(1000));
                mapper.ApplySave(System.Text.Json.JsonSerializer.Deserialize<SaveGameData>(json));
                foreach (var room in state.RoomStates)
                {
                    Equal(room.Slots[0].BotDurabilitySecondsRemaining, remaining, "old bot lifetime is not reset or rescaled");
                    Equal(room.Slots[0].BotRarity == rarity ? 1 : 0, 1, "old bot ownership retained");
                }
            }
        }
    }
    static void DurabilityTests()
    {
        foreach (double lifetime in new[] { 0.0, .000001, .5, 30, 100, 150, 40000 })
        foreach (long seconds in new long[] { 1, 30, 100, 150, 30L * 86400 })
        {
            var (s, e, o) = Create();
            var slot = Bot(lifetime); s.RoomStates[0].Slots[0] = slot;
            double rate = e.GetTotalIncomeWithoutTemporaryShopBoost();
            var save = new GameSaveService(s, e).CreateSave(1000);
            var result = o.ApplyFromSave(save, 1000 + seconds);
            Near(slot.BotDurabilitySecondsRemaining, Math.Max(0, lifetime - seconds), "remaining seconds");
            Equal(slot.BotBroken ? 1 : 0, seconds >= lifetime ? 1 : 0, "broken state");
            Near(result.OfflineEarned, rate * Math.Min(seconds, lifetime) * .25, "income ends at break");
            Equal(slot.IsRunning ? 1 : 0, seconds < lifetime ? 1 : 0, "automation state");
            // Round-trip actual main-save JSON and hydration, including broken state.
            var persisted = new GameSaveService(s, e).CreateSave(1000 + seconds);
            var json = System.Text.Json.JsonSerializer.Serialize(persisted);
            var (loaded, le, lo) = Create();
            new GameSaveService(loaded, le).ApplySave(System.Text.Json.JsonSerializer.Deserialize<SaveGameData>(json));
            Equal(loaded.RoomStates[0].Slots[0].BotDurabilitySecondsRemaining,
                slot.BotDurabilitySecondsRemaining, "JSON persistence");
            Equal(loaded.RoomStates[0].Slots[0].BotBroken ? 1 : 0, slot.BotBroken ? 1 : 0, "persist broken");
            Equal(lo.ApplyFromSave(persisted, persisted.LastSaveUnix).OfflineEarned, 0, "same timestamp");
        }
        // Independent Bots across rooms, locked rooms and slots, and upgraded machines.
        {
            var (s, e, o) = Create();
            s.RoomStates[0].Slots[0] = Bot(10);
            s.RoomStates[0].Slots.Add(Bot(50, BotRarity.Epic));
            var data = new RoomData("Second", null, null,
                [new MachineData("Base", 100, 1, 1, 100, null),
                 new MachineData("Upgrade", 500, 1, 1, 100, null)], 0);
            s.Rooms.Add(data);
            var room = new RoomState { Unlocked = true };
            var upgraded = Bot(100); upgraded.MachineTier = 1; upgraded.MachineLevel = 20;
            room.Slots.Add(upgraded); s.RoomStates.Add(room);
            double r0 = e.GetRoomIncome(0), r1 = e.GetRoomIncome(1);
            double firstRate = e.GetSlotIncome(0, s.RoomStates[0].Slots[0]);
            double expected = (firstRate * 10 + (r0 - firstRate) * 50 + r1 * 60) * .25;
            Near(o.ApplyFromSave(new GameSaveService(s, e).CreateSave(1000), 1060).OfflineEarned, expected, "multiple rooms/Bots");
            Equal(upgraded.BotDurabilitySecondsRemaining, 40, "upgraded Bot");
            room.Unlocked = false;
            var lockedSlot = Bot(100); lockedSlot.Unlocked = false;
            s.RoomStates[0].Slots.Add(lockedSlot);
            o.ApplyFromSave(new GameSaveService(s, e).CreateSave(1060), 1160);
            Equal(upgraded.BotDurabilitySecondsRemaining, 40, "locked room");
            Equal(lockedSlot.BotDurabilitySecondsRemaining, 100, "locked slot");
        }
        // Pipeline throughput is recalculated rather than subtracting a slot's
        // nominal output: bottlenecks can hide the first Bot's lost input.
        {
            var (s, e, o) = Create();
            s.Rooms.Add(s.Rooms[0]);
            var room = new RoomState { Unlocked = true };
            var first = Bot(10); var second = Bot(20);
            room.Slots.Add(first); room.Slots.Add(second); s.RoomStates.Add(room);
            double before = e.GetTotalIncomeWithoutTemporaryShopBoost();
            first.BotDurabilitySecondsRemaining = 0;
            double afterFirst = e.GetTotalIncomeWithoutTemporaryShopBoost();
            first.BotDurabilitySecondsRemaining = 10;
            var result = o.ApplyFromSave(new GameSaveService(s, e).CreateSave(1000), 1100);
            Near(result.OfflineEarned, (before * 10 + afterFirst * 10) * .25, "pipeline break recalculation");
            Equal(first.BotDurabilitySecondsRemaining + second.BotDurabilitySecondsRemaining, 0, "pipeline wear");
        }
        {
            var (s, e, o) = Create();
            var slot = new SlotData { Unlocked = true, BotRarity = BotRarity.Common };
            s.RoomStates[0].Slots[0] = slot;
            o.ApplyFromSave(Save(0), 1001);
            Equal(slot.BotDurabilitySecondsRemaining, slot.BotMaximumDurabilitySeconds - 1, "legacy initialization and zero-income wear");
        }
        // Online/offline equivalence at cycle boundaries, including exact break.
        foreach (int cycles in new[] { 1, 5, 10, 25, 30 })
        {
            var (a, ae, _) = Create(); var (b, be, offline) = Create();
            a.RoomStates[0].Slots[0] = Bot(100); b.RoomStates[0].Slots[0] = Bot(100);
            var production = new ProductionService(a, ae, new PipelineService(a, ae));
            double duration = ae.GetCycleDuration(a.RoomStates[0].Slots[0]);
            long seconds = (long)(cycles * duration);
            var save = new GameSaveService(b, be).CreateSave(1000);
            double earned = 0;
            for (int i = 0; i < seconds; i++) earned += production.Update(1);
            var result = offline.ApplyFromSave(save, 1000 + seconds);
            Near(result.OfflineEarned, earned * .25, "online/offline cycle-aligned income");
            Equal(b.RoomStates[0].Slots[0].BotDurabilitySecondsRemaining,
                a.RoomStates[0].Slots[0].BotDurabilitySecondsRemaining, "online/offline durability");
            Equal(b.RoomStates[0].Slots[0].BotBroken ? 1 : 0,
                a.RoomStates[0].Slots[0].BotBroken ? 1 : 0, "online/offline break");
            Near(b.RoomStates[0].Slots[0].CycleRemaining,
                a.RoomStates[0].Slots[0].CycleRemaining, "online/offline phase");
        }
        {
            var (s, e, o) = Create(); var slot = Bot(100); s.RoomStates[0].Slots[0] = slot;
            var production = new ProductionService(s, e, new PipelineService(s, e));
            production.Update(10);
            o.CapturePauseSnapshot(1000); o.CapturePauseSnapshot(1010);
            Equal(o.PauseUnix.Value, 1000, "pause save timestamp");
            o.ApplyResume(1030, out _);
            Equal(slot.BotDurabilitySecondsRemaining, 60, "runtime plus background once");
            Equal(o.ApplyResume(1060, out _) ? 1 : 0, 0, "repeat background resume");
            Equal(slot.BotDurabilitySecondsRemaining, 60, "no duplicate wear");
        }
        // Break boundaries combined with boost expiry and research completion.
        foreach (string id in new[] { ResearchCatalog.EfficientHardware1Id, ResearchCatalog.OfflineServersId })
        {
            var (s, e, o) = Create(); var slot = Bot(30); s.RoomStates[0].Slots[0] = slot;
            s.Lab.ActiveResearchId = id; s.Lab.ActiveResearchEndUnix = 1020;
            s.Shop.ProductionBoostEndUnix = 1010;
            var save = new GameSaveService(s, e).CreateSave(1000);
            double before = save.IncomePerSecond;
            new LabService(s, e).Update(1020);
            double after = e.GetTotalIncomeWithoutTemporaryShopBoost();
            double factor = .25 + s.Lab.GetOfflineIncomeBonus();
            s.Lab.LoadCompletedResearch([]); s.Lab.ActiveResearchId = id; s.Lab.ActiveResearchEndUnix = 1020;
            var result = o.ApplyFromSave(save, 1100);
            Near(result.OfflineEarned, Segment(before, 1000, 1020, 1010, .25) + after * 10 * factor, "research/boost/break");
            Equal(slot.BotDurabilitySecondsRemaining, 0, "research split wear");
        }
    }
    public static void Main()
    {
        ConfiguredLifetimeTests();
        DurabilityTests();
        // Includes zero/backward time, short/base, very long uncapped time, and all boost overlaps.
        foreach (long now in new long[] { 999, 1000, 1001, 1100, 1000 + 365L * 86400 })
        foreach (long boost in new long[] { 0, 999, 1050, now + 1 })
        foreach (int level in new int[] { 0, 4, 15 })
        {
            var (s, e, o) = Create(); s.Shop.OfflineUpgradeLevel = level;
            double expected = Segment(100, 1000, now, boost, Math.Clamp(.25 + level * .05, 0, 1));
            var r = o.ApplyFromSave(Save(100, boost), now);
            Equal(r.OfflineEarned, expected, "interval");
            Equal(s.Tokens, expected, "Tokens"); Equal(s.RunEarnedTokens, expected, "run");
            Equal(s.Stats.TotalEarned, expected, "lifetime"); Equal(s.Stats.OfflineEarned, expected, "offline stats");
        }
        foreach (string id in new[] { ResearchCatalog.EfficientHardware1Id, ResearchCatalog.OfflineServersId, "invalid" })
        foreach (long end in new long[] { 999, 1000, 1001, 1050, 1099, 1100, 1101 })
        foreach (long boost in new long[] { 0, 1025, 1050, 1075, 1200 })
        {
            var (s, e, o) = Create();
            double before = e.GetTotalIncomeWithoutTemporaryShopBoost();
            s.Lab.ActiveResearchId = id; s.Lab.ActiveResearchEndUnix = end;
            var r = o.ApplyFromSave(Save(before, boost), 1100);
            double after = e.GetTotalIncomeWithoutTemporaryShopBoost();
            double factor = Math.Clamp(.25 + s.Lab.GetOfflineIncomeBonus(), 0, 1);
            double expected = end > 1100 ? Segment(before, 1000, 1100, boost, .25)
                : end <= 1000 ? Segment(after, 1000, 1100, boost, factor)
                : Segment(before, 1000, end, boost, .25) + Segment(after, end, 1100, boost, factor);
            Equal(r.OfflineEarned, expected, "research split");
            Equal(r.ResearchResult.Changed ? 1 : 0, end <= 1100 ? 1 : 0, "research completion");
            Equal(s.Tokens, expected, "research credits");
        }
        foreach (int version in new[] { 1, 14, 15, 18, 19, 21 })
        {
            var (s, e, o) = Create();
            double income = version < 19 ? e.GetTotalIncomeWithoutTemporaryShopBoost() : 777;
            Equal(o.ApplyFromSave(Save(777, 1050, version), 1100).OfflineEarned,
                Segment(income, 1000, 1100, 1050, .25), "historical selection");
        }
        // Retained pre-v15 helper is currently bypassed by pre-v19 selection.
        var migration = typeof(OfflineProgressService).GetMethod("GetMigratedOfflineIncomePerSecond", BindingFlags.Static | BindingFlags.NonPublic);
        foreach (int version in new[] { 14, 15 })
        foreach (long boost in new long[] { 999, 1050 })
        foreach (double income in new double[] { 0, -1, 100 })
            Equal((double)migration.Invoke(null, new object[] { Save(income, boost, version) }),
                version < 15 && boost > 1000 && income > 0 ? income / GameConfig.ShopTemporaryProductionMultiplier : income, "legacy helper");
        foreach (long now in new long[] { 999, 1000, 1100 })
        {
            var (s, e, o) = Create();
            Equal(o.ApplyResume(now, out _) ? 1 : 0, 0, "no pause");
            s.Shop.ProductionBoostEndUnix = 1050;
            var pause = o.CapturePauseSnapshot(1000);
            Equal(o.CapturePauseSnapshot(1005).Unix, 1000, "duplicate pause");
            bool settled = o.ApplyResume(now, out var resume);
            Equal(settled ? 1 : 0, now > 1000 ? 1 : 0, "valid resume");
            var (_, _, cold) = Create();
            Equal(resume.OfflineEarned, cold.ApplyFromSave(Save(pause.BaseIncomePerSecond, 1050), now).OfflineEarned, "cold/resume");
            double tokens = s.Tokens;
            Equal(o.ApplyResume(1200, out _) ? 1 : 0, 0, "repeat resume");
            Equal(s.Tokens, tokens, "no double credit");
        }
        foreach (double income in new double[] { 0, -100 })
        {
            var (s, e, o) = Create();
            Equal(o.ApplyFromSave(Save(income), 1100).OfflineEarned, 0, "nonpositive income");
            Equal(s.Tokens, 0, "nonpositive credit");
        }
        foreach (long unix in new long[] { 0, -1 })
        {
            var (s, e, o) = Create(); o.CapturePauseSnapshot(unix);
            Equal(o.ApplyResume(1100, out _) ? 1 : 0, 0, "invalid snapshot");
            Equal(s.Tokens, 0, "invalid snapshot credit");
        }
        foreach (long now in new long[] { 999, 1000 })
        {
            var (s, e, o) = Create();
            s.Lab.ActiveResearchId = ResearchCatalog.EfficientHardware1Id;
            s.Lab.ActiveResearchEndUnix = now;
            var result = o.ApplyFromSave(Save(100), now);
            Equal(result.OfflineEarned, 0, "clock movement earnings");
            Equal(result.ResearchResult.Changed ? 1 : 0, 1, "clock movement research update");
        }
        Console.WriteLine($"PASS: {checks} deterministic offline assertions");
    }
}
