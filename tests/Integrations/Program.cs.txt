    static int checks;
    static void Eq<T>(T actual, T expected, string label)
    { if (!Equals(actual, expected)) throw new Exception($"{label}: {actual} != {expected}"); checks++; }
    static (GameState state, ShopService shop) Shop()
    {
        var state = new GameState(new List<RoomData>());
        var economy = new EconomyService(state);
        var progression = new ProgressionService(state, economy);

        return (state, new ShopService(state, economy, production));

    sealed class Ad : IRewardedAdHandle
    {
        public Action Earned, Dismissed;
        public Action<string> Failed;
        public int Shows, Destroys;
        public bool Throw;
        public void Show(Action earned, Action dismissed, Action<string> failed)
        { Shows++; Earned = earned; Dismissed = dismissed; Failed = failed; if (Throw) throw new Exception("show"); }
        public void Destroy() { Destroys++; }
    }
    sealed class Ads : IRewardedAdAdapter
    {
        public int Initializations, Loads;
        public Action<IRewardedAdHandle> Loaded;
        public Action<string> Failed;
        public bool Throw;
        public void Initialize() { Initializations++; }
        public void Load(Action<IRewardedAdHandle> loaded, Action<string> failed)
        { Loads++; Loaded = loaded; Failed = failed; if (Throw) throw new Exception("load"); }
    }

    static void Purchases()
    {
        foreach (var product in IapCatalog.GetAll())
        foreach (int quantity in new[] { -1, 0, 1, 3 })
        {
            var (state, shop) = Shop(); var order = new List<string>(); int completions = 0; string message = "";
            var workflow = new IapGrantWorkflow(amount => { order.Add("grant"); return shop.GrantDataShards(amount); },
                _ => false, _ => { order.Add("save"); return true; }, _ => { order.Add("complete"); completions++; return true; });
            workflow.MessageRequested += value => message = value;
            workflow.Handle(product.ProductId, "token", quantity); workflow.Handle(product.ProductId, "token", quantity);
            Eq(state.Shop.DataShards, (double)(product.ShardAmount * Math.Max(1, quantity)), "catalog/quantity reward");
            Eq(string.Join(",", order), "grant,save,complete", "financial order"); Eq(completions, 1, "complete once");
            Eq(message, $"+{product.ShardAmount * Math.Max(1, quantity)} Data Shards!", "purchase message");
        }
        int grants = 0, saves = 0, complete = 0;
        var unknown = new IapGrantWorkflow(_ => { grants++; return new(true, "ok"); }, _ => false,
            _ => { saves++; return true; }, _ => { complete++; return true; });
        unknown.Handle("unknown", "token", 1); unknown.Handle("data_shard_100", "", 1);
        unknown.Handle("data_shard_5500", "overflow", int.MaxValue);
        Eq(grants, 0, "invalid product/token/overflow no grant"); Eq(saves, 0, "invalid no save"); Eq(complete, 0, "invalid no complete");
        var processed = new IapGrantWorkflow(_ => { grants++; return new(true, "ok"); }, _ => true, _ => true, _ => true);
        processed.Handle("data_shard_100", "processed", 1); Eq(grants, 0, "processed ledger token no grant");
        int aborted = 0;
        var failed = new IapGrantWorkflow(_ => new(false, "fail"), _ => false, _ => { saves++; return true; },
            _ => { complete++; return true; }, abort: () => aborted++);
        failed.Handle("data_shard_100", "failed", 1); Eq(saves, 0, "failed grant no save"); Eq(complete, 0, "failed grant no complete"); Eq(aborted, 1, "failed grant cancels reservation");

        var retry = new IapGrantWorkflow(_ => { grants++; return new(true, "+100 Data Shards!"); }, _ => false,
            _ => { saves++; return saveOk; }, _ => { complete++; return completeOk; });
        retry.Handle("data_shard_100", "retry", 1); Eq(grants, 1, "failed save retains one grant"); Eq(complete, 0, "failed save no complete");

            { boundary.Add("grant"); reentrant.Handle("data_shard_100", "reentrant", 1); return new(true, "+100 Data Shards!"); },
            _ => false, _ => { boundary.Add("save"); reentrant.RetryPending(); return true; },
            _ => { boundary.Add("complete"); return true; },

    static void Sync()
    {
        var calls = new List<string>(); long level = 151, prestige = 3; long firebaseLevel = 0, firebasePrestige = 0;
        var workflow = new PlayGamesSyncWorkflow(() => calls.Add("achievements"),
            () => { calls.Add("official level " + level); calls.Add("official prestige " + prestige); },
            () => { firebaseLevel = level; firebasePrestige = prestige; calls.Add("firebase"); }, _ => throw new Exception("unexpected"));
        using var service = new PlayGamesSyncService(); service.Initialize(workflow); service.Initialize(workflow);
        Eq(service.GetChildCount(), 1, "one timer registration"); Eq(service.GetChild<Godot.Timer>(0).WaitTime, 1d, "one-second cadence");
        service.Tick(); Eq(string.Join(",", calls), "achievements,official level 151,official prestige 3,firebase", "one tick fanout");
        Eq(firebaseLevel, level, "Firebase same level"); Eq(firebasePrestige, prestige, "Firebase same prestige");
        calls.Clear(); level = 152; prestige = 4; service.Tick(); Eq(calls.Count, 4, "next tick reads fresh scores"); Eq(firebasePrestige, 4L, "Prestige updates");
        int errors = 0, forwarded = 0;
        new PlayGamesSyncWorkflow(() => throw new Exception("unavailable"), () => forwarded++, () => forwarded++, _ => errors++).Tick();
        Eq(errors, 1, "unavailable fails safely"); Eq(forwarded, 2, "independent sync continues");
        using var uninitialized = new PlayGamesSyncService(); uninitialized.Tick(); Eq(uninitialized.Initialized, false, "uninitialized safe");
        service.Free(); uninitialized.Free();
    }

    static void BillingLedger()
    {
        string ledger = ProjectSettings.GlobalizePath("user://idle_ai_iap_ledger.json");
        string temporary = ProjectSettings.GlobalizePath("user://idle_ai_iap_ledger.tmp.json");
        Directory.CreateDirectory(Path.GetDirectoryName(ledger));
        File.WriteAllText(ledger, "{\"ProcessedPurchaseTokens\":[\"old-token\"]}");
        var billing = new GooglePlayBillingService();
        Eq(billing.IsPurchaseProcessed("old-token"), true, "existing ledger compatible");
        Eq(billing.CompleteConsumableGrant("new-token"), true, "ledger success boundary");
        var restarted = new GooglePlayBillingService();
        Eq(restarted.IsPurchaseProcessed("new-token"), true, "token survives restart");
        Eq(restarted.IsPurchaseProcessed("old-token"), true, "existing token retained");
        Eq(File.Exists(temporary), false, "temporary ledger replaced");
        Directory.CreateDirectory(temporary);
        Eq(restarted.CompleteConsumableGrant("failed-token"), false, "ledger failure refuses completion");
        Eq(restarted.IsPurchaseProcessed("failed-token"), false, "failed token not marked successful");
        Eq(new GooglePlayBillingService().IsPurchaseProcessed("failed-token"), false, "failed token not persisted");
        Directory.Delete(temporary);
        foreach (string corrupt in new[] { "broken", "null", "{}", "{\"ProcessedPurchaseTokens\":null}", "{\"ProcessedPurchaseTokens\":[null]}" })
        {
            File.WriteAllText(ledger, corrupt);
            var locked = new GooglePlayBillingService();
            Eq(locked.LedgerLoadFailed, true, "corrupt ledger locks billing");
            Eq(locked.CompleteConsumableGrant("unsafe-token"), false, "corrupt ledger cannot finalize");
            Eq(File.ReadAllText(ledger), corrupt, "corrupt ledger preserved");
        }
        File.Delete(ledger);
    }
    static void MainSaveRecovery()
    {
        string main = ProjectSettings.GlobalizePath("user://idle_ai_save.json");
        string backup = ProjectSettings.GlobalizePath("user://idle_ai_save.bak.json");
        using var manager = new SaveManager();
        var data = new SaveGameData { SaveVersion = 21, Tokens = 123 };
        data.Rooms.Add(new RoomSaveData { Unlocked = true });
        data.Rooms[0].Slots.Add(new SlotSaveData { Unlocked = true });
        Eq(manager.SaveData(data), true, "baseline save");
        Eq(manager.SaveData(data), true, "baseline backup");
        foreach (string malformed in new[] { "{\"Rooms\":null}", "{\"Rooms\":[null]}", "{\"Rooms\":[{\"Slots\":null}]}", "{\"Rooms\":[{\"Slots\":[null]}]}" })
        {
            File.WriteAllText(main, malformed);
            Eq(manager.LoadData()?.Tokens, 123d, "structural corruption recovers backup");
            Eq(manager.LastLoadFailed, false, "backup recovery unlocks saves");
        }
        File.WriteAllText(main, "{\"Rooms\":null}");
        File.WriteAllText(backup, "broken");
        Eq(manager.LoadData(), null, "both corrupt rejected");
        Eq(manager.LastLoadFailed, true, "both corrupt lock saves");
        Eq(manager.SaveData(data), false, "corruption cannot overwrite");
        Eq(File.ReadAllText(main), "{\"Rooms\":null}", "main evidence preserved");
        Eq(File.ReadAllText(backup), "broken", "backup evidence preserved");
        File.Delete(main); File.Delete(backup); manager.Free();
    }
