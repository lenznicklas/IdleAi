using Godot;
using IdleAi;
using System;
using System.Linq;
using System.Text.Json;

public static class Program
{
    private const string Path = "user://singularity_save.json";
    private static int _assertions;
    private static long _now = 1_000_000;
    private static readonly JsonSerializerOptions Json = new() { WriteIndented = true };
    private static void Check(bool value, string name)
    {
        _assertions++;
        if (!value) throw new Exception(name);
    }
    private static void Equal(double actual, double expected, string name) =>
        Check(Math.Abs(actual - expected) <= Math.Max(1e-12, Math.Abs(expected) * 1e-12), $"{name}: {actual:R} != {expected:R}");
    private static void Write(SingularitySaveData data)
    {
        using var file = Godot.FileAccess.Open(Path, Godot.FileAccess.ModeFlags.Write);
        file.StoreString(JsonSerializer.Serialize(data, Json));
    }
    private static SingularitySaveData Read()
    {
        using var file = Godot.FileAccess.Open(Path, Godot.FileAccess.ModeFlags.Read);
        return JsonSerializer.Deserialize<SingularitySaveData>(file.GetAsText())!;
    }
    private static void Clear() => DirAccess.RemoveAbsolute(ProjectSettings.GlobalizePath(Path));
    private static SingularityService Service(GameState? state = null, double roll = 0) => new(state, () => _now, () => roll);
    private static GameState Normal()
    {
        var state = new GameState([]);
        for (int i = 0; i < 4; i++)
        {
            var room = new RoomState { Unlocked = true };
            for (int n = 0; n < 8; n++) room.Slots.Add(new SlotData { Unlocked = true });
            state.RoomStates.Add(room);
        }
        return state;
    }
    private sealed class Domain
    {
        internal readonly SingularitySaveData Data = new() { Unlocked = true };
        internal readonly SingularityEconomy Economy;
        internal readonly SingularitySectorService Sectors;
        internal readonly SingularityBotService Bots;
        internal readonly SingularityNodeService Nodes;
        internal readonly SingularityProductionService Production;
        internal readonly SingularityPersistenceService Persistence;
        internal readonly SingularityOfflineService Offline;
        internal int Notifications, Saves, Rolls;
        internal double Roll;
        internal Domain(LabData? lab = null)
        {
            Economy = new(Data);
            Sectors = new(Data, Economy, Notify, () => Saves++);
            Bots = new(Data, Sectors, Economy, Notify, lab, () => { Rolls++; return Roll; });
            Nodes = new(Data, Sectors, Economy, Notify);
            Production = new(Data, Sectors, Bots);
            Persistence = new(Data, Sectors, Bots, () => _now);
            Offline = new(Data, Bots, Production, Persistence, Economy, () => Notifications++, () => _now);
            Sectors.EnsureSector(0);
        }
        private void Notify() { Saves++; Notifications++; }
        internal SingularityNodeData Node(int sector, int node, SingularityNodeType type, int level = 1)
        {
            var result = Sectors.GetNode(sector, node);
            result.Type = type; result.Level = level;
            return result;
        }
    }

        Clear();
        Console.WriteLine($"Singularity: {_assertions} assertions passed.");
    }

    private static void DefaultsAndUnlock()
    {
        Clear(); var state = Normal(); var service = Service(state);
        Check(!service.Unlocked, "default locked"); Equal(service.Matter, 0, "default Matter");
        Check(service.SectorCount == 1 && service.CurrentSectorIndex == 0 && service.CoreCount == 1, "default sector/core");
        Check(service.GetSector(0).Nodes.Count == 8 && service.CoreLevel == 1, "default nodes/core level");
        Equal(service.GetTotalOutputPerSecond(), 0, "locked output");
        service.Update(10); Equal(service.Matter, 0, "locked Update");
        Check(!service.Unlock(state).Changed, "insufficient Tokens"); Equal(state.Tokens, 0, "failed unlock Tokens");
        state.Tokens = SingularityService.TestUnlockTokenCost;
        state.RoomStates[3].Slots[7].Unlocked = false;
        Check(!service.Unlock(state).Changed, "normal progression prerequisite"); Equal(state.Tokens, SingularityService.TestUnlockTokenCost, "prerequisite no spend");
        state.RoomStates[3].Slots[7].Unlocked = true;
        int events = 0; service.Changed += () => events++;
        var result = service.Unlock(state, new EconomyService(state));
        Check(result.Changed && result.Message == "SINGULARITY ONLINE • SECTOR 1 CORE INSTALLED", "unlock/message");
        Equal(state.Tokens, 0, "unlock spends exactly once"); Equal(state.Stats.TotalSpent, SingularityService.TestUnlockTokenCost, "unlock stats");
        Check(service.Unlocked && events == 1, "unlock notifies once");
        Check(!service.Unlock(state).Changed && events == 1, "duplicate unlock");
        Equal(service.Matter, 0, "normal unlock leaves Matter");
        Equal(service.GetTotalOutputPerSecond(), .05, "first core output");
    }

    private static void Nodes()
    {
        var d=new Domain();
        Check(d.Nodes.IsNodeTypeUnlocked(SingularityNodeType.Compute)&&d.Nodes.IsNodeTypeUnlocked(SingularityNodeType.Amplifier),"immediate types");
        Check(!d.Nodes.IsNodeTypeUnlocked(SingularityNodeType.Cooling)&&!d.Nodes.IsNodeTypeUnlocked(SingularityNodeType.Quantum),"locked types");
        for(int i=1;i<5;i++)
        {
            d.Sectors.GetSector(i).CoreUnlocked=true;
            Check(d.Nodes.IsNodeTypeUnlocked(SingularityNodeType.Cooling)==(i>=2),"Cooling requires 3 cores");
            Check(d.Nodes.IsNodeTypeUnlocked(SingularityNodeType.Quantum)==(i>=4),"Quantum requires 5 cores");
        }
        Check(!d.Nodes.BuildNode(0,0,SingularityNodeType.Compute).Changed,"insufficient Matter"); Equal(d.Data.Matter,0,"failed spend");
        d.Data.Matter=10; Check(d.Nodes.BuildNode(0,0,SingularityNodeType.Compute).Changed,"Compute build"); Equal(d.Data.Matter,0,"exact build spend");
        Equal(d.Sectors.GetNode(0,0).InvestedMatter,10,"build investment");
        Equal(d.Nodes.GetBuildCost(0,SingularityNodeType.Amplifier),45*1.22,"global growth");
        Equal(d.Nodes.GetBuildCost(2,SingularityNodeType.Compute),10*9*1.22,"sector cost scaling");
        d.Data.Matter=10000;
        Check(d.Nodes.BuildNode(0,1,SingularityNodeType.Amplifier).Changed,"Amplifier build");
        Check(d.Nodes.BuildNode(0,2,SingularityNodeType.Cooling).Changed,"Cooling build");
        Check(d.Nodes.BuildNode(0,3,SingularityNodeType.Quantum).Changed,"Quantum build");
        Check(!d.Nodes.BuildNode(0,0,SingularityNodeType.Compute).Changed,"occupied no spend");
        double before=d.Data.Matter;
        Check(d.Nodes.UpgradeNode(0,0).Changed,"node upgrade"); Equal(before-d.Data.Matter,18,"upgrade exactly once");
        Equal(d.Sectors.GetNode(0,0).InvestedMatter,28,"upgrade investment"); Equal(d.Nodes.GetNodeUpgradeCost(0,0),18*1.85,"node upgrade progression");
        Equal(d.Nodes.GetNodeSellRefund(0,0),28.0/3,"node refund"); before=d.Data.Matter;
        Check(d.Nodes.SellNode(0,0).Changed,"node sale"); Equal(d.Data.Matter,before+28.0/3,"refund credited");
        Check(d.Sectors.GetNode(0,0).Type==SingularityNodeType.Empty&&d.Sectors.GetNode(0,0).InvestedMatter==0,"sale clears state");
        Check(!d.Nodes.SellNode(0,0).Changed,"duplicate sale");
        var legacy=d.Node(0,4,SingularityNodeType.Compute,3);
        Equal(d.Nodes.GetNodeSellRefund(0,4),(10+18+18*1.85)/3,"legacy investment estimate");
        Check(d.Nodes.UpgradeNode(0,4).Changed,"legacy upgrade"); Equal(legacy.InvestedMatter,10+18+18*1.85+18*Math.Pow(1.85,2),"legacy upgrade investment");
    }

    private static void ConfiguredBotLifetimes()
    {
        foreach (var (rarity, hours, oldHours) in new[]
        {
            (BotRarity.Common, 24, 6), (BotRarity.Rare, 36, 12),
            (BotRarity.Epic, 48, 24), (BotRarity.Legendary, 72, 48)
        })
        {
            var d = new Domain(); var node = d.Node(0, 0, SingularityNodeType.Compute);
            node.BotRarity = rarity; node.BotPurchasePriceMatter = 100;
            d.Bots.InitializeNodeBotDurability(node);
            double lifetime = hours * 3600;
            Equal(node.BotDurabilitySecondsRemaining, lifetime, "Singularity configured lifetime: " + rarity);
            d.Bots.ConsumeSingularityBotWork(60);
            Equal(node.BotDurabilitySecondsRemaining, lifetime - 60, "Singularity runtime wear");
            d.Offline.ApplyOfflineIncomeFromTimestamp(_now, _now + 3600);
            Equal(node.BotDurabilitySecondsRemaining, lifetime - 3660, "Singularity offline wear");
            d.Offline.ApplyOfflineIncomeFromTimestamp(_now + 3600, _now + 3600 + (long)lifetime);
            Check(node.BotBroken, "Singularity lifetime reaches zero");
            Equal(d.Bots.GetNodeBotEffectiveMultiplier(0, 0), 1, "broken Singularity bot contributes no bonus");
            d.Data.Matter = 1000;
            Equal(d.Bots.GetNodeBotRepairCost(0, 0), 20, "Singularity repair price unchanged");
            Check(d.Bots.RepairNodeBot(0, 0).Changed, "Singularity repair succeeds");
            Equal(node.BotDurabilitySecondsRemaining, lifetime, "Singularity repair restores new lifetime");
            Check(!d.Bots.RepairNodeBot(0, 0).Changed, "Singularity full bot cannot be repaired twice");
            foreach (double remaining in new[] { 0, 1234.5, oldHours * 3600.0 })
            {
                node.BotDurabilitySecondsRemaining = remaining;
                d.Persistence.Save(); var loaded = new Domain(); loaded.Persistence.Load();
                var restored = loaded.Sectors.GetNode(0, 0);
                Equal(restored.BotDurabilitySecondsRemaining, remaining, "Singularity old save seconds preserved");
                Check(restored.BotRarity == rarity, "Singularity old bot retained");
            }
        }
    }
